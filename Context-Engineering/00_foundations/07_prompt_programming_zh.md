## 核心思想：认知操作即函数

提示编程的根本洞见是将认知操作视为可调用的函数：

```
┌─────────────────────────────────────────────────────────────────────┐
│ 传统提示                        │ 提示编程                          │
├──────────────────────────────────┼──────────────────────────────────┤
│ “分析一战的原因，考虑政治、      │ analyze(                          │
│ 经济和社会因素。”               │   topic="causes of World War I", │
│                                 │   factors=["political",          │
│                                 │            "economic",           │
│                                 │            "social"],            │
│                                 │   depth="comprehensive",         │
│                                 │   format="structured"            │
│                                 │ )                                 │
└──────────────────────────────────┴──────────────────────────────────┘
```

两种方式都能得到类似结果，但提示编程版本：
1. 参数显式
2. 输入可系统变换
3. 模板可复用
4. 明确引导模型推理结构

## 提示编程的实际实现

让我们看看提示编程的实际用法：

### 1. 基本函数定义与调用

```
# 定义认知函数
function summarize(text, length="short", style="informative", focus=null) {
  // 函数描述
  // 用指定参数总结文本
  if (length not in ["short", "medium", "long"]) {
    throw Error("Length 必须为 short、medium 或 long");
  }
  summary_length = {
    "short": "1-2段",
    "medium": "3-4段",
    "long": "5段及以上"
  }[length];
  focus_instruction = focus ? 
    `重点关注${focus}相关内容。` : 
    "均衡覆盖所有要点。";
  return `
    任务：总结下列文本。
    参数：
    - 长度：${summary_length}
    - 风格：${style}
    - 特别说明：${focus_instruction}
    待总结文本：
    ${text}
    请用${style}风格、${summary_length}总结文本。
    ${focus_instruction}
  `;
}

# 调用函数
input_text = "关于气候变化的长文...";
summarize(input_text, length="medium", focus="经济影响");
```

### 2. 函数组合

```
# 定义多个认知函数
function research(topic, depth="comprehensive", sources=5) {
  return `调研${topic}，深度：${depth}，参考${sources}个来源。`;
}
function analyze(information, framework="thematic", perspective="neutral") {
  return `用${framework}框架、${perspective}视角分析如下信息：${information}`;
}
function synthesize(analysis, format="essay", tone="academic") {
  return `将如下分析整合为${format}，语气：${tone}：${analysis}`;
}
# 组合实现复杂任务
research_results = research("人工智能对就业的影响", depth="detailed", sources=8);
analysis_results = analyze(research_results, framework="因果", perspective="平衡");
final_output = synthesize(analysis_results, format="报告", tone="专业");
```

### 3. 条件逻辑与控制流

```
function solve_math_problem(problem, show_work=true, check_solution=true) {
  if contains_variables(problem) {
    approach = "代数";
    steps = ["识别变量和常数", "列方程", "求解未知数", "验证原题"];
  } else if contains_geometry_terms(problem) {
    approach = "几何";
    steps = ["识别几何属性", "应用公式", "计算所需值", "验证解的合理性"];
  } else {
    approach = "算术";
    steps = ["分步计算", "按顺序运算", "得出结果", "验证计算"];
  }
  prompt = `
    任务：解决如下${approach}问题。
    问题：${problem}
    ${show_work ? "请分步展示解题过程：" : "只需给出最终答案。"}
    ${show_work ? steps.map((step, i) => `${i+1}. ${step}`).join("\n") : ""}
    ${check_solution ? "解完后请验证答案是否满足原题所有条件。" : ""}
  `;
  return prompt;
}
```
